# SSH Command (Latest Instance)
ssh -J ubuntu@3.107.1.103 ec2-user@10.0.21.113

# DEBUG HEALTH CHECK FAILURE
# Once connected, run these commands IN ORDER:

# A. Check what target group is checking
echo "=== Target Group Health Check Settings ==="
aws elbv2 describe-target-groups --region ap-southeast-2 --query "TargetGroups[?contains(TargetGroupName, 'app')][TargetGroupName,Port,HealthCheckPort,HealthCheckPath,HealthCheckProtocol]" --output table

# B. Check Security Group Rules
echo "=== Security Group Ingress Rules ==="
INSTANCE_ID=$(ec2-metadata --instance-id | cut -d " " -f 2)
SG_ID=$(aws ec2 describe-instances --instance-ids $INSTANCE_ID --region ap-southeast-2 --query 'Reservations[0].Instances[0].SecurityGroups[0].GroupId' --output text)
aws ec2 describe-security-groups --group-ids $SG_ID --region ap-southeast-2 --query 'SecurityGroups[0].IpPermissions' --output table

# C. Get Internal ALB IP addresses
echo "=== Internal ALB IPs (these IPs should be allowed to connect) ==="
aws ec2 describe-network-interfaces --region ap-southeast-2 --filters "Name=description,Values=*internal-alb*" --query 'NetworkInterfaces[*].PrivateIpAddress' --output table

# Once connected, run these commands:

# 1. Check Docker container status
docker ps -a

# 2. Check container logs (last 100 lines)
docker logs ecommerce-backend --tail 100

# 3. Check if container is listening on port 8080
netstat -tlnp | grep 8080

# 4. Test health endpoint from inside instance
curl -v http://localhost:8080/health

# 5. Check user-data execution logs
sudo tail -100 /var/log/cloud-init-output.log

# 6. Check if ECR login succeeded
grep -i "login succeeded" /var/log/cloud-init-output.log

# 7. Check environment variables passed to container
docker inspect ecommerce-backend | grep -A 20 Env

# 8. Test database connection from instance
nc -zv ${DB_HOST} 3306

# 9. Check if image was pulled successfully
docker images | grep ecommerce-backend

# 10. If container not running, try to start it manually
docker start ecommerce-backend

# 11. Check target group health check from ALB perspective
curl -v http://localhost:8080/health -H "User-Agent: ELB-HealthChecker/2.0"

# 12. Check which port is actually being health checked
echo "=== Checking netstat for active connections ==="
sudo netstat -tnp | grep ESTABLISHED | grep docker

# 13. Test if ALB can reach this instance on port 8080
echo "=== Testing from different IPs/ports ==="
# Listen for incoming connections
sudo tcpdump -i any port 8080 -n -c 20

# 14. Check CloudWatch logs for health check requests
docker logs ecommerce-backend --tail 50 | grep -E "GET /health|GET /health.txt"

# 15. Simulate ALB health check exactly as it does
echo "=== Simulating ALB Health Check ==="
timeout 5 curl -v -H "User-Agent: ELB-HealthChecker/2.0" http://localhost:8080/health 2>&1 | grep -E "HTTP|Connected|status"
